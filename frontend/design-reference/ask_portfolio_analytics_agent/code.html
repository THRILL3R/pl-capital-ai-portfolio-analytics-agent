SELECT
  s.ticker,
  s.name,
  h.shares,
  h.cost_basis,
  s.current_price,
  (h.shares * s.current_price - h.cost_basis) as unrealized_pnl
FROM holdings h
JOIN securities s ON h.security_id = s.id
JOIN portfolios p ON h.portfolio_id = p.id
WHERE p.name = ?
ORDER BY h.cost_basis DESC
LIMIT 5;
